ALTER PROCEDURE [dbo].[GetArbitratorsReport]
    @EmployeeIds NVARCHAR(MAX) = NULL,              -- Filter by specific arbitrator employee IDs
    @FromDate DATETIME = NULL,                      -- Start date for filtering
    @ToDate DATETIME = NULL,                        -- End date for filtering
    @ArbitratorName NVARCHAR(200) = NULL,           -- Filter by arbitrator name (partial match)
    @ArbitratorStatus NVARCHAR(50) = NULL,          -- Filter by arbitrator status (Active/Inactive)
    @ExperienceYears INT = NULL,                    -- Filter by years of experience
    @Skip INT = 0,                                  -- Number of records to skip (for pagination)
    @Take INT = NULL                                -- Number of records to take (NULL = all records)
AS
BEGIN
    -- Prevent row count messages from being returned to improve performance
    SET NOCOUNT ON;
    
    -- Validate pagination parameters
    SET @Skip = ISNULL(@Skip, 0);
    IF @Skip < 0 SET @Skip = 0;
    
    -- Set default date range if not provided
    -- Default from date: earliest possible date to include all historical records
    SET @FromDate = ISNULL(@FromDate, '1900-01-01');
    -- Default to date: current date and time to include all records up to now
    SET @ToDate = ISNULL(@ToDate, GETDATE());

    -- =============================================
    -- CTE 1: AllArbitrators
    -- Purpose: Get ALL arbitrator citizens first, regardless of litigation assignment
    -- This ensures we show all arbitrators even those with no cases
    -- =============================================
    ;WITH AllArbitrators AS (
        SELECT DISTINCT
            C.Id AS ArbitratorCitizenId,
            C.FullName AS ArbitratorName,
            -- Convert boolean ArbitratorStatusId to descriptive text
            CASE WHEN C.ArbitratorStatusId = 1 THEN N'مفعل' ELSE N'غير مفعل' END AS ArbitratorStatus,
            ISNULL(CR.ExperienceYears, 0) AS ExperienceYears
        FROM Citizen C
        LEFT JOIN CitizenRequest CR ON C.Id = CR.CitizenId
        WHERE 
            -- Get ALL citizens who are arbitrators (based on IsArbitrator flag)
            C.IsArbitrator = 1
            -- Apply arbitrator-specific filters
            AND (@EmployeeIds IS NULL OR C.Id IN (
                SELECT value FROM STRING_SPLIT(@EmployeeIds, ',')
            ))
            -- Filter by arbitrator name (partial match, case-insensitive)
            AND (@ArbitratorName IS NULL OR C.FullName LIKE '%' + @ArbitratorName + '%')
            -- Filter by arbitrator status with proper boolean conversion
            AND (@ArbitratorStatus IS NULL OR 
                 (@ArbitratorStatus = 1 AND C.ArbitratorStatusId = 1) OR 
                 (@ArbitratorStatus = 2 AND C.ArbitratorStatusId = 2))
            -- Filter by years of experience
            AND (@ExperienceYears IS NULL OR ISNULL(CR.ExperienceYears, 0) = @ExperienceYears)
            

    ),
    
    -- =============================================
    -- CTE 2: ChairmanLitigations
    -- Purpose: Get litigation requests where arbitrator is the chairman/main arbitrator
    -- =============================================
    ChairmanLitigations AS (
        SELECT 
            A.ArbitratorCitizenId,
            L.Id,
            L.IsFinished,
            L.LitigationRequestStage,
            L.ArbitrationPeriod
        FROM AllArbitrators A
        -- LEFT JOIN to keep ALL arbitrators even those without litigation cases
        LEFT JOIN (
            SELECT
                L2.*,
                ROW_NUMBER() OVER (PARTITION BY L2.Id ORDER BY L2.CreatedOn DESC) AS rn
            FROM LitigationRequest AS L2
            WHERE
                L2.ArbitratorCitizenId IS NOT NULL
                AND L2.LitigationRequestStage = 3       -- Only arbitration stage cases
                AND L2.SentOn >= @FromDate              -- Filter by date range
                AND L2.SentOn <= @ToDate
        ) AS L ON L.ArbitratorCitizenId = A.ArbitratorCitizenId AND L.rn = 1
    ),
    
    -- =============================================
    -- CTE 3: RegularArbitratorLitigations
    -- Purpose: Get litigation requests where arbitrator is a regular arbitrator (panel member)
    -- =============================================
    RegularArbitratorLitigations AS (
        SELECT 
            A.ArbitratorCitizenId,
            L.Id,
            L.IsFinished,
            L.LitigationRequestStage,
            L.ArbitrationPeriod
        FROM AllArbitrators A
        -- Left join with litigation requests where this person is a regular arbitrator
        LEFT JOIN LitigationRequestArbitrator LRA ON LRA.ArbitratorCitizenId = A.ArbitratorCitizenId
        LEFT JOIN LitigationRequest L ON L.Id = LRA.LitigationRequestId 
                                      AND L.LitigationRequestStage = 3    -- Only arbitration stage cases
                                      AND L.SentOn >= @FromDate           -- Filter by date range
                                      AND L.SentOn <= @ToDate
    ),
    
    -- =============================================
    -- CTE 4: CombinedLitigations
    -- Purpose: Combine both chairman and regular arbitrator litigations
    -- =============================================
    CombinedLitigations AS (
    SELECT DISTINCT 
        ArbitratorCitizenId,
        Id,
        IsFinished,
        LitigationRequestStage,
        ArbitrationPeriod
    FROM (
        SELECT ArbitratorCitizenId, Id, IsFinished, LitigationRequestStage, ArbitrationPeriod
        FROM ChairmanLitigations
        UNION ALL
        SELECT ArbitratorCitizenId, Id, IsFinished, LitigationRequestStage, ArbitrationPeriod
        FROM RegularArbitratorLitigations
    ) AS X
),
    
    -- =============================================
    -- CTE 5: BaseArbitratorLitigations
    -- Purpose: Join all arbitrators with their combined litigation statistics
    -- =============================================
    BaseArbitratorLitigations AS (
        SELECT 
            A.ArbitratorCitizenId,
            A.ArbitratorName,
            A.ArbitratorStatus,
            A.ExperienceYears,
            CL.Id,
            CL.IsFinished,
            CL.LitigationRequestStage,
            CL.ArbitrationPeriod
        FROM AllArbitrators A
        LEFT JOIN CombinedLitigations CL ON CL.ArbitratorCitizenId = A.ArbitratorCitizenId
    )
    
    -- =============================================
    -- MAIN QUERY: Generate aggregated report by arbitrator
    -- Purpose: Group litigation data by arbitrator and calculate key statistics
    -- Includes both chairman and regular arbitrator cases
    -- =============================================
    SELECT
        -- Arbitrator identification
        ArbitratorName,
        ArbitratorStatus,
        ExperienceYears,
        
        -- Core count metrics (counts ALL cases: chairman + regular arbitrator)
        COUNT(Id) AS TotalLitigationCount,                          -- Total cases assigned to this arbitrator
        
        -- Completion status counts (following same pattern as reviewers/reformers)
        SUM(CASE WHEN IsFinished = 0 THEN 1 ELSE 0 END) AS NotFinishedCount,    -- Cases not finished
        SUM(CASE WHEN IsFinished = 1 THEN 1 ELSE 0 END) AS FinishedCount,       -- Cases finished
        
        -- Delayed cases (cases in stage 3 that have exceeded arbitration period)
        SUM(CASE 
            WHEN LitigationRequestStage = 3                 -- In arbitration stage
                 AND ArbitrationPeriod = 0                  -- Period has expired
            THEN 1 ELSE 0 
        END) AS DueDateCount,                                       -- Overdue arbitration cases
        
        -- Performance ratios (percentages)
        -- Calculate percentage of cases that are not finished
        CAST(
            SUM(CASE WHEN IsFinished = 0 THEN 1 ELSE 0 END) * 100.0 / 
            NULLIF(COUNT(Id), 0)                                    -- Avoid division by zero
            AS DECIMAL(5,2)
        ) AS NotFinishedRatio,                                      -- Percentage of pending cases
        
        -- Calculate percentage of cases that are finished
        CAST(
            SUM(CASE WHEN IsFinished = 1 THEN 1 ELSE 0 END) * 100.0 / 
            NULLIF(COUNT(Id), 0)                                    -- Avoid division by zero
            AS DECIMAL(5,2)
        ) AS FinishedRatio,                                         -- Percentage of completed cases
        
        -- Calculate percentage of delayed cases
        CAST(
            SUM(CASE 
                WHEN LitigationRequestStage = 3 AND ArbitrationPeriod = 0 
                THEN 1 ELSE 0 
            END) * 100.0 / NULLIF(COUNT(Id), 0)                     -- Avoid division by zero
            AS DECIMAL(5,2)
        ) AS DueDateRatio,                                          -- Percentage of overdue arbitration cases
        
        -- Total count for pagination info (total records before pagination)
        COUNT(*) OVER() AS TotalCount
        
    FROM BaseArbitratorLitigations
    -- Group by arbitrator to get per-arbitrator statistics
    -- Note: We group by ALL arbitrators, including those with zero cases
    GROUP BY ArbitratorName, ArbitratorStatus, ExperienceYears
    ORDER BY ArbitratorName                                         -- Sort results alphabetically by arbitrator name
    
    -- =============================================
    -- PAGINATION: Apply OFFSET and FETCH NEXT for pagination
    -- =============================================
    OFFSET @Skip ROWS                                               -- Skip the specified number of rows
    FETCH NEXT 
        CASE 
            WHEN @Take IS NULL THEN 2147483647                      -- If @Take is NULL, fetch maximum possible rows
            ELSE @Take 
        END ROWS ONLY                                               -- Fetch only the specified number of rows
END

    @FromDate DATETIME = NULL,                      -- Start date for filtering
    @ToDate DATETIME = NULL,                        -- End date for filtering
    @ReformerName NVARCHAR(200) = NULL,             -- Filter by reformer name (partial match)
    @ReformerStatus INT = NULL,                     -- Filter by reformer status (Active/Inactive)
    @ExperienceYears INT = NULL,                    -- Filter by years of experience
    @Skip INT = 0,                                  -- Number of records to skip (for pagination)
    @Take INT = NULL                                -- Number of records to take (NULL = all records)
AS
BEGIN
    -- Prevent row count messages from being returned to improve performance
    SET NOCOUNT ON;
    
    -- Validate pagination parameters
    SET @Skip = ISNULL(@Skip, 0);
    IF @Skip < 0 SET @Skip = 0;
    
    -- Set default date range if not provided
    SET @FromDate = ISNULL(@FromDate, '1900-01-01');
    SET @ToDate = ISNULL(@ToDate, GETDATE());

    -- =============================================
    -- CTE 1: AllReformers - Get ALL reformer citizens
    -- Purpose: Start with ALL reformers first, regardless of litigation assignment
    -- =============================================
    WITH AllReformers AS (
        SELECT DISTINCT
            C.Id AS ReformerCitizenId,
            C.FullName AS ReformerName,
            CASE WHEN C.ReformerStatusId = 1 THEN N'مفعل' ELSE N'غير مفعل' END AS ReformerStatus,
            ISNULL(CR.ExperienceYears, 0) AS ExperienceYears
        FROM Citizen C
        LEFT JOIN CitizenRequest CR ON C.Id = CR.CitizenId
        WHERE 
            -- Get ALL citizens who are reformers (based on IsReformer flag)
            C.IsReformer = 1
            
            -- Apply reformer-specific filters
            AND (@ReformerName IS NULL OR C.FullName LIKE '%' + @ReformerName + '%')
            AND (@ReformerStatus IS NULL OR 
                 (@ReformerStatus = 1 AND C.ReformerStatusId = 1) OR 
                 (@ReformerStatus = 2 AND C.ReformerStatusId = 2))
            AND (@ExperienceYears IS NULL OR ISNULL(CR.ExperienceYears, 0) = @ExperienceYears)
    ),
    
    -- =============================================
    -- CTE 2: ReformerLitigationStats - Get litigation statistics for each reformer
    -- Purpose: Calculate statistics for litigation requests assigned to each reformer
    -- =============================================
    ReformerLitigationStats AS (
        SELECT 
            R.ReformerCitizenId,
            R.ReformerName,
            R.ReformerStatus,
            R.ExperienceYears,
            
            -- Reformer litigation statistics
            COUNT(L.Id) AS TotalLitigationCount,
            SUM(CASE WHEN L.IsFinished = 0 THEN 1 ELSE 0 END) AS NotFinishedCount,
            SUM(CASE WHEN L.IsFinished = 1 THEN 1 ELSE 0 END) AS FinishedCount,
            SUM(CASE WHEN L.LitigationRequestStage = 2 AND L.ReconciliationPeriod = 0 THEN 1 ELSE 0 END) AS DueDateCount
            
        FROM AllReformers R
        
        -- Left join with litigation requests where this person is the reformer
        LEFT JOIN (
            SELECT L.*,
                   ROW_NUMBER() OVER (PARTITION BY L.Id ORDER BY L.CreatedOn DESC) AS rn
            FROM LitigationRequest L
            WHERE L.ReformerCitizenId IS NOT NULL
              AND L.LitigationRequestStage = 2
              AND L.SentOn >= @FromDate 
              AND L.SentOn <= @ToDate
        ) L ON L.ReformerCitizenId = R.ReformerCitizenId AND L.rn = 1
        
        GROUP BY R.ReformerCitizenId, R.ReformerName, R.ReformerStatus, R.ExperienceYears
    )
    
    -- =============================================
    -- MAIN QUERY: Generate final reformer report with statistics
    -- =============================================
    SELECT
        ReformerName,
        ReformerStatus,
        ExperienceYears,
        
        -- Core count metrics
        ISNULL(TotalLitigationCount, 0) AS TotalLitigationCount,
        ISNULL(NotFinishedCount, 0) AS NotFinishedCount,
        ISNULL(FinishedCount, 0) AS FinishedCount,
        ISNULL(DueDateCount, 0) AS DueDateCount,
        
        -- Performance ratios (percentages)
        CAST(
            ISNULL(NotFinishedCount, 0) * 100.0 / 
            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(5,2)
        ) AS NotFinishedRatio,
        
        CAST(
            ISNULL(FinishedCount, 0) * 100.0 / 
            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(5,2)
        ) AS FinishedRatio,
        
        CAST(
            ISNULL(DueDateCount, 0) * 100.0 / 
            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(5,2)
        ) AS DueDateRatio
		,COUNT(*) OVER() AS TotalCount
        
    FROM ReformerLitigationStats
    -- Show ALL reformers, even those with no litigation cases (they'll have zero statistics)
    ORDER BY ReformerName
    
    -- =============================================
    -- PAGINATION: Apply OFFSET and FETCH NEXT for pagination
    -- =============================================
    OFFSET @Skip ROWS
    FETCH NEXT 
        CASE 
            WHEN @Take IS NULL THEN 2147483647
            ELSE @Take 
        END ROWS ONLY
END

              AND L.LitigationRequestStage = 2
        ) L ON L.ReformerCitizenId = R.PersonnelId AND L.rn = 1 AND R.PersonnelType = N'مصلح'
        WHERE R.PersonnelType = N'مصلح'
        GROUP BY R.PersonnelId, R.PersonnelName, R.PersonnelType, R.PersonnelStatus, R.ExperienceYears , R.PersonnelTypeId
    ),
    
    -- =============================================
    -- CTE 7: ArbitratorStats - Calculate litigation statistics for arbitrators
    -- =============================================
    ArbitratorStats AS (

    SELECT
        A.PersonnelId,
        A.PersonnelName,
        A.PersonnelType,
        A.PersonnelTypeId,
        A.PersonnelStatus, -- Corrected from ArbitratorStatus
        A.ExperienceYears,

        -- unified counts (no double counting)
        COUNT(DISTINCT AL.LitigationId) AS TotalLitigationCount,

        COUNT(DISTINCT CASE WHEN AL.IsFinished = 0 THEN AL.LitigationId END) AS NotFinishedCount,
        COUNT(DISTINCT CASE WHEN AL.IsFinished = 1 THEN AL.LitigationId END) AS FinishedCount,
        COUNT(DISTINCT CASE WHEN AL.LitigationRequestStage = 3 AND AL.ArbitrationPeriod = 0 THEN AL.LitigationId END)
            AS DueDateCount,

        -- chairman-only counts (no duplication)
        COUNT(DISTINCT CASE WHEN AL.IsChairman = 1 THEN AL.LitigationId END) AS ChairmanTotalCount,
        COUNT(DISTINCT CASE WHEN AL.IsChairman = 1 AND AL.IsFinished = 0 THEN AL.LitigationId END) AS ChairmanNotFinishedCount,
        COUNT(DISTINCT CASE WHEN AL.IsChairman = 1 AND AL.IsFinished = 1 THEN AL.LitigationId END) AS ChairmanFinishedCount,
        COUNT(DISTINCT CASE WHEN AL.IsChairman = 1 AND AL.LitigationRequestStage = 3 AND AL.ArbitrationPeriod = 0 THEN AL.LitigationId END)
            AS ChairmanDueDateCount,

        -- regular arbitrator-only counts (no duplication)
        COUNT(DISTINCT CASE WHEN AL.IsRegular = 1 THEN AL.LitigationId END) AS ArbitratorTotalCount,
        COUNT(DISTINCT CASE WHEN AL.IsRegular = 1 AND AL.IsFinished = 0 THEN AL.LitigationId END) AS ArbitratorNotFinishedCount,
        COUNT(DISTINCT CASE WHEN AL.IsRegular = 1 AND AL.IsFinished = 1 THEN AL.LitigationId END) AS ArbitratorFinishedCount,

        A.PersonnelStatus, A.ExperienceYears

    -- =============================================
    -- CTE 8: UnifiedStats - Union all statistics
    -- =============================================
    UnifiedStats AS (
        SELECT * FROM ReviewerStats
        UNION ALL
        SELECT * FROM ReformerStats
        UNION ALL
        SELECT * FROM ArbitratorStats
        UNION ALL
        -- Include personnel with no statistics (zero counts)
        SELECT 
            AP.PersonnelId,
            AP.PersonnelName,
            AP.PersonnelType,
			AP.PersonnelTypeId,
            AP.PersonnelStatus,
            AP.ExperienceYears,
            0 AS TotalLitigationCount,
            0 AS NotFinishedCount,
            0 AS FinishedCount,
            0 AS DueDateCount,
            0 AS ChairmanTotalCount,
            0 AS ChairmanNotFinishedCount,
            0 AS ChairmanFinishedCount,
            0 AS ChairmanDueDateCount,
            0 AS ArbitratorTotalCount,
            0 AS ArbitratorNotFinishedCount,
            0 AS ArbitratorFinishedCount,
            0 AS ArbitratorDueDateCount
        FROM AllPersonnel AP
        WHERE NOT EXISTS (

            UNION
            SELECT 1 FROM ReformerStats RF WHERE RF.PersonnelId = AP.PersonnelId
            UNION
            SELECT 1 FROM ArbitratorStats AR WHERE AR.PersonnelId = AP.PersonnelId
        )

    )
    
    -- =============================================
    -- MAIN QUERY: Generate final unified personnel report
    -- =============================================
    SELECT
        PersonnelName,
        PersonnelType,
		PersonnelTypeId,
        PersonnelStatus,
        ExperienceYears,
        
        -- Core count metrics
        ISNULL(TotalLitigationCount, 0) AS TotalLitigationCount,
        ISNULL(NotFinishedCount, 0) AS NotFinishedCount,
        ISNULL(FinishedCount, 0) AS FinishedCount,
        ISNULL(DueDateCount, 0) AS DueDateCount,
        
        -- Arbitrator-specific metrics (will be 0 for non-arbitrators)
        ISNULL(ChairmanTotalCount, 0) AS ChairmanTotalCount,
        ISNULL(ChairmanNotFinishedCount, 0) AS ChairmanNotFinishedCount,
        ISNULL(ChairmanFinishedCount, 0) AS ChairmanFinishedCount,
        ISNULL(ChairmanDueDateCount, 0) AS ChairmanDueDateCount,
        ISNULL(ArbitratorTotalCount, 0) AS ArbitratorTotalCount,
        ISNULL(ArbitratorNotFinishedCount, 0) AS ArbitratorNotFinishedCount,
        ISNULL(ArbitratorFinishedCount, 0) AS ArbitratorFinishedCount,
        ISNULL(ArbitratorDueDateCount, 0) AS ArbitratorDueDateCount,
        
        -- Performance ratios (percentages)
        CAST(

            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(18,2) -- Increased precision to avoid arithmetic overflow error (Error 8115)
        ) AS NotFinishedRatio,
        
        CAST(

            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(18,2) -- Increased precision to avoid arithmetic overflow error (Error 8115)
        ) AS FinishedRatio,
        
        CAST(
            ISNULL(DueDateCount, 0) * 100.0 / 
            NULLIF(ISNULL(TotalLitigationCount, 0), 0)
            AS DECIMAL(18,2) -- Increased precision to avoid arithmetic overflow error (Error 8115)
        ) AS DueDateRatio
        ,
		COUNT(*) OVER() AS TotalCount
    FROM UnifiedStats
	Where 
	(@PersonnelTypeId IS NULL OR PersonnelTypeId in (SELECT value FROM STRING_SPLIT(@PersonnelTypeId, ',')))

    ORDER BY PersonnelType, PersonnelName
    
    -- =============================================
    -- PAGINATION: Apply OFFSET and FETCH NEXT for pagination
    -- =============================================
    OFFSET @Skip ROWS
    FETCH NEXT 
        CASE 
            WHEN @Take IS NULL THEN 2147483647
            ELSE @Take 
        END ROWS ONLY
END
